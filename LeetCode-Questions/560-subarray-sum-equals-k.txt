class Solution {
    public int subarraySum(int[] nums, int k) {
        
        int prefix = 0, answer = 0;

        Map<Integer, Integer> map = new HashMap<>();

        map.put(0, 1);

        for(int n : nums){

            prefix += n;

            int needed = prefix - k;

            if(map.containsKey(needed)){
                answer += map.get(needed);
            }

            map.put(prefix, map.getOrDefault(prefix, 0) + 1);
        }

        return answer;
    }
}
