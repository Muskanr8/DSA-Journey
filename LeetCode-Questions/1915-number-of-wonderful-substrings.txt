class Solution {
    public long wonderfulSubstrings(String word) {
        
        HashMap<Integer, Integer> map = new HashMap<>();
        map.put(0, 1);

        long answer = 0;
        int mask = 0;

        for(int i = 0; i < word.length(); i++){

            char ch = word.charAt(i);

            switch(ch){

                case 'a' : mask ^= (1 << 0); break;
                case 'b' : mask ^= (1 << 1); break;
                case 'c' : mask ^= (1 << 2); break;
                case 'd' : mask ^= (1 << 3); break;
                case 'e' : mask ^= (1 << 4); break;
                case 'f' : mask ^= (1 << 5); break;
                case 'g' : mask ^= (1 << 6); break;
                case 'h' : mask ^= (1 << 7); break;
                case 'i' : mask ^= (1 << 8); break;
                case 'j' : mask ^= (1 << 9); break;
            }

            // Same mask → 0 odd-frequency characters
            answer += map.getOrDefault(mask, 0);

            // One bit different → 1 odd-frequency character
            for(int bit = 0; bit < 10; bit++){

                int newMask = mask ^ (1 << bit);

                answer += map.getOrDefault(newMask, 0);
            }

            // Update frequency AFTER checking
            map.put(mask, map.getOrDefault(mask, 0) + 1);
        }

        return answer;
    }
}
